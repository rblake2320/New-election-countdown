import React from 'react';
import {readFileSync,appendFileSync} from 'node:fs';
import {join} from 'node:path';
import {createHash} from 'node:crypto';
import {beforeEach,afterEach,it,expect,vi} from 'vitest';
import {render,screen,within,cleanup} from '@testing-library/react';
import {GeneralRoster} from '../client/src/components/general-roster';
const boundary=vi.hoisted(()=>({data:undefined as any,isLoading:false,error:null as Error|null}));
vi.mock('../client/src/hooks/use-general-roster',()=>({useGeneralRoster:()=>boundary}));
const bytes=readFileSync(join(__dirname,'fixtures/phase2-ma3-eaa3008/MA-52.json'));
const digest=createHash('sha256').update(bytes).digest('hex');
if(digest!=='4a22f9821004f35ba8443fb19053a589b3cff437d46a5deea29af54e7980198b')throw new Error('Real MA3 payload bytes changed');
const capture=JSON.parse(bytes.toString('utf8'));
if(capture.receipt.httpStatus!==200||capture.receipt.manifestSha256!=='eaa3008ea0589aab351d41fb203a45361d64fc983d4ac3e8c344cfbd528e9bb5'||capture.body.status!=='sourced')throw new Error('Expected root actual private PostgreSQL/HTTP MA3 capture');
const deniedFetch=vi.fn(()=>{throw new Error('Network forbidden in offline receiving review');});
beforeEach(()=>{Object.assign(boundary,{data:capture.body,isLoading:false,error:null});deniedFetch.mockClear();vi.stubGlobal('fetch',deniedFetch);});
afterEach(()=>{expect(deniedFetch).not.toHaveBeenCalled();cleanup();vi.unstubAllGlobals();});
function mount(){const view=render(<GeneralRoster electionId={52}/>);const observation={state:'MA3',payloadSha256:digest,candidateCount:capture.body.candidates.length,renderedText:view.container.textContent,sourceLinks:[...view.container.querySelectorAll('a')].map(a=>({text:a.textContent,href:a.getAttribute('href')})),componentFetchCalls:deniedFetch.mock.calls.length};if(process.env.PHASE2_RECEIVING_OBSERVATION_FILE)appendFileSync(process.env.PHASE2_RECEIVING_OBSERVATION_FILE,JSON.stringify(observation)+'\n');return view;}
function rows(){return within(screen.getByRole('list',{name:'Candidates on the official general-election list'})).getAllByRole('listitem');}
it('MA3 all three actual printed ticket names reach unique receiving rows',()=>{mount();expect(rows().map(r=>r.querySelector('span.font-medium')?.textContent)).toEqual(['James and Kittredge','Healey and Driscoll','Minogue and Oliver']);expect(rows().length).toBe(3);});
it('MA3 James uses current General Independent party rather than historical finance party',()=>{mount();const c=capture.body.candidates.find((c:any)=>c.name==='James and Kittredge');expect(c.party).toBe('I');expect(c.generalListSource.partyRaw).toBe('Independent');expect(c.generalListSource.legalName).toBe('Andrea Corinne James');const row=rows().find(r=>r.querySelector('span.font-medium')?.textContent===c.name)!;expect(row.textContent).toBe('James and KittredgeIndependent');expect(row.textContent).not.toContain('Democratic');});
it('MA3 both distinct identity-only documents reach correctly labelled safe links',()=>{mount();const evidence=screen.getByTestId('general-roster-evidence');const links=within(evidence).getAllByRole('link');expect(links.length).toBe(2);for(const ev of new Map(capture.body.candidates.map((c:any)=>[c.generalListSource.identityEvidence.url,c.generalListSource.identityEvidence])).values() as any){const link=within(evidence).getByRole('link',{name:ev.title});expect(link.getAttribute('href')).toBe(ev.url);expect(link.getAttribute('rel')).toBe('noopener noreferrer');expect(link.parentElement?.textContent).toContain('Name identity source:');}expect(evidence.textContent).toContain('(name identity only)');expect(evidence.textContent).toContain('(name and Governor role only)');expect(evidence.textContent).toContain('(p. 1)');});
it('MA3 printed-only scope appears without an invented complete write-in roster',()=>{mount();expect(capture.body.coverage).toBe('scoped');expect(capture.body.sources).toHaveLength(1);expect(capture.body.sources[0].scope).toEqual({office:'G',district:'00',expectedCount:3,ballotAccessType:'printed'});const note=screen.getByTestId('general-roster-scope-note');expect(note.textContent).toContain('printed-ballot candidates');expect(note.textContent).not.toContain('write-in');expect(screen.queryAllByText('Write-in')).toHaveLength(0);});
it('MA3 General eligibility source and checked date stay distinct from identity evidence dates',()=>{mount();const sources=screen.getByTestId('general-roster-sources');const links=within(sources).getAllByRole('link');expect(links.length).toBe(1);expect(links[0].getAttribute('href')).toBe('https://www.sec.state.ma.us/divisions/elections/research-and-statistics/2026-state-election-candidates.htm');expect(sources.textContent).toContain('checked Oct 11, 2026');expect(sources.textContent).not.toContain('published');expect(sources.textContent).not.toContain('2025');expect(sources.textContent).not.toContain('primary');expect(sources.textContent).not.toContain('OCPF');});
it('MA3 no source identity proof is promoted to ballot-results certification or unknown incumbency',()=>{mount();for(const c of capture.body.candidates){expect(c.ballotStatus).toBe('not_certified');expect(c.generalListSource.ballotCertificationPromoted).toBe(false);expect(c.isIncumbent).toBeNull();}expect(screen.queryAllByText('Incumbent')).toHaveLength(0);expect(screen.getByTestId('general-roster').textContent).toContain('Being listed is not the same as certification of results.');});
