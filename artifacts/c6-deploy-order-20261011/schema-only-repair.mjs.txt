import {createRequire} from 'node:module';
import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
const require=createRequire('D:/projects/et-wt/sol-congress-reviewed-main/package.json');
const pg=require('pg'),dotenv=require('dotenv');
const connectionString=dotenv.parse(readFileSync('D:/projects/New-election-countdown/.env')).DATABASE_URL;
const target=new URL(connectionString);
if(!['localhost','127.0.0.1'].includes(target.hostname)||(target.port||'5432')!=='5432'||target.pathname!=='/elections')throw Error('Approved election DB guard refused');
const sourceHead='0a717fecf838b5f8ab5c56805519537d3621ef28';
const migration=execFileSync('git',['-C','D:/projects/et-wt/sol-congress-reviewed-main','show',sourceHead+':migrations/20261011_official_results_routes.sql']);
const hash=v=>createHash('sha256').update(v).digest('hex');
const body=migration.toString('utf8').replace(/^BEGIN;\s*$/m,'').replace(/^COMMIT;\s*$/m,'');
if(/\b(?:INSERT|UPDATE|DELETE|TRUNCATE|DROP)\b/i.test(body.replace(/--[^\n]*/g,'')))throw Error('Reviewed migration unexpectedly contains data mutation');
const expected=['official_results_url','official_results_source'];
const receipt={operation:'schema-only',sourceHead,migrationSha256:hash(migration),sourceOrPollValueImport:false,committed:false};
const pool=new pg.Pool({connectionString,max:1,connectionTimeoutMillis:10000,allowExitOnIdle:true});
const client=await pool.connect();
const columns=async()=> (await client.query("SELECT column_name,data_type,is_nullable,column_default FROM information_schema.columns WHERE table_schema='public' AND table_name='elections' AND column_name=ANY($1) ORDER BY column_name",[expected])).rows;
const snapshot=async(exclude)=>{
  const elections=(await client.query("SELECT count(*)::int AS count,coalesce(jsonb_agg(jsonb_build_object('id',id,'xmin',xmin::text,'row',to_jsonb(e)-$1::text[]) ORDER BY id),'[]'::jsonb)::text AS bytes FROM public.elections e",[exclude])).rows[0];
  const candidates=(await client.query("SELECT count(*)::int AS count,coalesce(jsonb_agg(jsonb_build_object('id',id,'xmin',xmin::text,'row',to_jsonb(c)) ORDER BY id),'[]'::jsonb)::text AS bytes FROM public.candidates c")).rows[0];
  return {elections:elections.count,electionOldFieldsAndXminSha256:hash(elections.bytes),candidates:candidates.count,candidateFullRowsAndXminSha256:hash(candidates.bytes)};
};
try {
  await client.query('BEGIN ISOLATION LEVEL SERIALIZABLE');
  await client.query("SET LOCAL timezone='UTC'; SET LOCAL lock_timeout='5s'; SET LOCAL statement_timeout='30s'");
  await client.query('SELECT pg_advisory_xact_lock($1)',[2026110301]);
  await client.query('LOCK TABLE public.elections,public.candidates IN SHARE ROW EXCLUSIVE MODE');
  const existing=await columns();
  const added=expected.filter(name=>!existing.some(col=>col.column_name===name));
  receipt.addedColumns=added;
  receipt.before=await snapshot(added);
  if(receipt.before.elections!==537||receipt.before.candidates!==136)throw Error('Expected held catalog counts changed; no repair applied');
  await client.query(body);
  receipt.columns=await columns();
  if(receipt.columns.length!==2||receipt.columns.some(c=>c.is_nullable!=='YES'||c.column_default!==null||c.data_type!==(c.column_name==='official_results_url'?'text':'jsonb')))throw Error('Reviewed nullable metadata shape mismatch');
  receipt.after=await snapshot(added);
  if(JSON.stringify(receipt.before)!==JSON.stringify(receipt.after))throw Error('Existing rows or row versions changed; rollback');
  receipt.auditRows=(await client.query('SELECT count(*)::int AS count FROM public.official_results_sync_runs')).rows[0].count;
  if(receipt.auditRows!==0)throw Error('Unexpected shared results data operation; rollback');
  for(const name of added){if((await client.query(`SELECT count(*)::int AS count FROM public.elections WHERE ${name} IS NOT NULL`)).rows[0].count!==0)throw Error('Added metadata contains values');}
  await client.query(body);
  receipt.repeat=await snapshot(added);
  if(JSON.stringify(receipt.after)!==JSON.stringify(receipt.repeat))throw Error('Repeat migration changed existing rows');
  await client.query('COMMIT');receipt.committed=true;
  await client.query("SET timezone='UTC'");
  receipt.fresh=await snapshot(added);
  if(JSON.stringify(receipt.after)!==JSON.stringify(receipt.fresh))throw Error('Postcommit readback mismatch');
  receipt.mainQuery=(await client.query('SELECT id,official_results_url,official_results_source FROM public.elections WHERE id=37')).rows;
  receipt.outcome='Worked';
} catch(error) {
  if(!receipt.committed)await client.query('ROLLBACK');
  receipt.outcome=receipt.committed?'Applied; verification failed':'Failed; rolled back';receipt.errorCode=error.code??error.name;receipt.reason=error.code?'Database operation refused':error.message;
  process.exitCode=1;
} finally {client.release();await pool.end();writeFileSync('C:/Users/techai/.codex/tmp/sol-c6-schema-only-repair.json',JSON.stringify(receipt,null,2));}
