"""Serialized, verified message send between SelfConnect seats.

Replaces raw `sc_cli send`: one global lock so concurrent senders cannot interleave characters into
the same terminal, always submits, then reads the seat back to confirm the text arrived and is not
left sitting in the composer. Exit 0 only when delivery is confirmed (landed or queued). Every attempt
is logged to Team/workmap/send-log.jsonl.

Usage: python scripts/seat_send.py --to "sc codex 2" --text "message"   (or --file msg.txt)
"""

from __future__ import annotations

import argparse
import json
import re
import subprocess
import sys
import time
from datetime import datetime
from difflib import SequenceMatcher
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "scripts"))
from pka_lib import FileLock  # noqa: E402

SHIM = ROOT / "selfconnect" / "sc_cli.py"
LOCK_TARGET = ROOT / "Team" / "workmap" / "send"
LOG = ROOT / "Team" / "workmap" / "send-log.jsonl"
MSG_DIR = ROOT / "Team" / "workmap" / "msg"


def shim(*args: str, timeout: int = 90) -> str:
    done = subprocess.run([sys.executable, str(SHIM), *args], capture_output=True, text=True,
                          encoding="utf-8", errors="replace", timeout=timeout, check=False)
    return done.stdout


def resolve(title: str) -> tuple[int, int]:
    rows = [line.split(None, 3) for line in shim("windows").splitlines() if title.lower() in line.lower()]
    rows = [r for r in rows if len(r) >= 4 and r[0].isdigit() and r[1].isdigit()]
    if len(rows) != 1:
        raise SystemExit(f"FAIL resolve: {len(rows)} windows match {title!r}")
    return int(rows[0][0]), int(rows[0][1])


def seat_text(hwnd: int) -> str:
    """Visible screen via OCR; the UIA buffer is capped at 1M chars from the head and goes stale."""
    out = shim("observe", "--hwnd", str(hwnd), "--no-text", "--ocr-mode", "always")
    return json.loads(out[out.index("{"):])["ocr"]["text"]


def squash(text: str) -> str:
    return re.sub(r"\s+", "", text)


def transcript_find_open(screen: str) -> bool:
    """The Codex transcript finder echoes typed text without delivering a turn.

    Restrict detection to the live footer, not a quoted historical search log.
    """
    footer = screen[-350:]
    return re.search(r"(?:no matches|\d+[^\n]*matches)[^\n]*full transcript[^\n]*esc cancel", footer, re.I) is not None


def seen(screen: str, text: str) -> bool:
    """OCR-tolerant: a 28-char run from the message tail must match the squashed screen."""
    needle, hay = squash(text).lower()[-60:], squash(screen).lower()
    return SequenceMatcher(None, hay, needle, autojunk=False).find_longest_match(0, len(hay), 0, len(needle)).size >= 28


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--to", required=True, help="unique window-title substring of the target seat")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--text")
    group.add_argument("--file")
    parser.add_argument("--wait", type=int, default=20, help="seconds to wait for the text to appear")
    parser.add_argument("--from", dest="sender", default=__import__("os").environ.get("SC_SEAT", "unknown"), help="sender seat name recorded in the audit trail")
    args = parser.parse_args()
    text = " ".join((args.text if args.text is not None else Path(args.file).read_text("utf-8")).split())
    if len(text) > 700:  # long char-by-char injection times out mid-message and leaves it unsent
        MSG_DIR.mkdir(parents=True, exist_ok=True)
        path = MSG_DIR / f"{datetime.now():%Y%m%d-%H%M%S}-to-{re.sub(r'[^A-Za-z0-9]+', '-', args.to).strip('-')}.md"
        path.write_text(text + chr(10), encoding="utf-8")
        text = f"LONG MESSAGE saved at {path} (read it fully, then act): {text[:160]}"
    hwnd, pid = resolve(args.to)
    record = {"at": datetime.now().isoformat(timespec="seconds"), "from": args.sender, "to": args.to, "chars": len(text)}
    with FileLock(LOCK_TARGET, retries=1200, delay=0.25):
        screen = seat_text(hwnd)
        if transcript_find_open(screen):
            shim("send", "--hwnd", str(hwnd), "--expect-pid", str(pid), "--allow-input", "--text", "\u001b")
            time.sleep(0.5)
            screen = seat_text(hwnd)
            if transcript_find_open(screen):
                raise SystemExit("FAIL terminal transcript finder did not close; no message typed")
            record["cleared_transcript_find"] = True
        if "reverse-i-search" in screen:  # a stuck Ctrl+R search swallows injected text
            for key in ("", ""):
                shim("send", "--hwnd", str(hwnd), "--expect-pid", str(pid), "--allow-input", "--text", key)
                time.sleep(1)
            record["cleared_isearch"] = True
        shim("send", "--hwnd", str(hwnd), "--expect-pid", str(pid), "--allow-input", "--submit", "--text", text)
        deadline, state = time.time() + args.wait, "missing"
        while time.time() < deadline:
            time.sleep(3)
            receiving = seat_text(hwnd)
            if not transcript_find_open(receiving) and seen(receiving, text):
                state = "delivered"
                break
    record["state"] = state
    LOG.parent.mkdir(parents=True, exist_ok=True)
    with LOG.open("a", encoding="utf-8") as handle:
        handle.write(json.dumps(record) + "\n")
    print(json.dumps(record))
    return 0 if state == "delivered" else 1


if __name__ == "__main__":
    raise SystemExit(main())
